# Arrival alerts: "tell me when Alice is within 500 m of W1" and "Bob within 1 km of me".
# Positions arrive with the console (cpos = as if shared over the radio). Once = removed after
# it fires; repeat = fires again only after leaving (25 % further) and coming back.
# Run: run-script.bat scripts\geofence.txt cmap
gps on
expect 60000 [BOOT] ready
wait 2000
serial geof clear
serial contact 0000000000000000000000000000a11c Alice
serial contact 0000000000000000000000000000b0bb Bob
serial wpt clear
serial wpt add 43.6600 -79.3700 PARK
expect 3000 [WPT] W1 PARK
serial geof add 0000000000000000000000000000a11c w1 500 0
expect 3000 [GEOF] add ok
serial geof add 0000000000000000000000000000b0bb me 1000 1
expect 3000 [GEOF] add ok
# Alice 2 km away: nothing. Then 300 m from PARK: the alert, and it is gone (once).
serial cpos 0000000000000000000000000000a11c 43.6800 -79.3700 1
expect 3000 [LOC] Alice at
serial cpos 0000000000000000000000000000a11c 43.6625 -79.3700 0
expect 3000 [GEOF] Alice is
wait 1500
shot geof-01-alert.png
serial geof
expect 3000 [GEOF] 1 alerts
# Bob near me (the sim fix 43.6532,-79.3832): fires, then not again until he left and came back
serial cpos 0000000000000000000000000000b0bb 43.6560 -79.3832 0
expect 3000 [GEOF] Bob is
serial geof
expect 3000 (inside)
serial cpos 0000000000000000000000000000b0bb 43.6700 -79.3832 0
expect 3000 [LOC] Bob at
serial cpos 0000000000000000000000000000b0bb 43.6540 -79.3832 0
expect 3000 [GEOF] Bob is
# The settings pages
key M
wait 600
key S
wait 600
key 9
wait 600
key 5
wait 800
shot geof-02-list.png
key 2
wait 800
shot geof-03-new.png
key BACK
wait 400
key BACK
wait 400
key BACK
wait 400
key BACK
wait 400
serial geof clear
quit
