# Map + GPS per power mode, and the Nav keys card (K, or the middle tab, on the Atlas).
# Balanced: opening the map asks for a GPS visit. Saver: it does not (last fix + age shown).
# Custom: only with "Map wakes GPS" on (pset mapgps 1). Tiles: build\simdata\cmap\sd.
# Run: run-script.bat scripts\map_keys.txt cmap
gps off
usb off
expect 60000 [BOOT] ready
# Balanced (the mode change starts a visit; indoors it ends after 60 s)
serial pm1
expect 3000 [POWER] Balanced
expect 70000 (no fix), next in 15 min
key T
wait 1500
key A
expect 3000 [GPS] visit requested (map)
wait 1500
serial map close
expect 3000 [MAP] closed
# Saver: the map does not wake the GPS
serial pm2
expect 3000 [POWER] Saver
expect 70000 (no fix), next in 60 min
key T
wait 1500
key A
expect 3000 [GPS] map: no visit in this power mode
wait 7000
shot keys-01-saver.png
serial map close
expect 3000 [MAP] closed
# Custom with "Map wakes GPS" on
serial pset mapgps 1
expect 3000 [POWER] Custom
expect 70000 (no fix), next in 60 min
key T
wait 1500
key A
expect 3000 [GPS] visit requested (map)
wait 1500
# K shows the Nav keys card (the map stays usable above it); map keys work with it showing
key K
expect 3000 [NAV] keys shown
wait 1500
shot keys-02-navkeys.png
key Z
wait 1200
key T
wait 1200
serial map
expect 3000 m *
key D
wait 1200
# K again hides it; ^ hides it too (the map stays open)
key K
expect 3000 [NAV] keys hidden
key K
expect 3000 [NAV] keys shown
key BACK
expect 3000 [NAV] keys hidden
serial map
expect 3000 [MAP] open z
# M on the Atlas = the normal MENU; I leaves the map for the Inbox
key M
wait 1500
shot keys-03-menu.png
key BACK
wait 800
key I
expect 3000 [MAP] closed
wait 1000
# back to Balanced for later scripts
serial pm1
expect 3000 [POWER] Balanced
quit
