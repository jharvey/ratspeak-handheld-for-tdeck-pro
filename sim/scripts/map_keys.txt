# Map + GPS per power mode, and the map keys menu (M on the Atlas).
# Balanced: opening the map asks for a GPS visit. Saver: it does not (last fix + age shown).
# Custom: only with "Map wakes GPS" on (pset mapgps 1). M on the map: the keys as a
# pull-down (G/V + Enter, or the key itself). Tiles: build\simdata\cmap\sd.
# Run: run-script.bat scripts\map_keys.txt cmap
gps off
usb off
expect 60000 [BOOT] ready
# Balanced (the mode change starts a visit; indoors it ends after 60 s)
serial pm1
expect 3000 [POWER] Balanced
expect 70000 (no fix), next in 15 min
key G
wait 1500
key L
expect 3000 [GPS] visit requested (map)
wait 1500
serial map close
expect 3000 [MAP] closed
# Saver: the map does not wake the GPS
serial pm2
expect 3000 [POWER] Saver
expect 70000 (no fix), next in 60 min
key G
wait 1500
key L
expect 3000 [GPS] map: no visit in this power mode
wait 7000
shot keys-01-saver.png
serial map close
expect 3000 [MAP] closed
# Custom with "Map wakes GPS" on
serial pset mapgps 1
expect 3000 [POWER] Custom
expect 70000 (no fix), next in 60 min
key G
wait 1500
key L
expect 3000 [GPS] visit requested (map)
wait 1500
# The keys menu: M, then V V Enter = Zoom out
key M
expect 3000 [MAP] keys menu
wait 1500
shot keys-02-menu.png
key V
wait 300
key V
wait 300
key ENTER
expect 3000 [MAP] menu: Zoom out
wait 1500
# M then ^ closes the menu only (the map stays)
key M
expect 3000 [MAP] keys menu
key BACK
wait 1500
serial map
expect 3000 [MAP] open z
# M then M = main menu
key M
expect 3000 [MAP] keys menu
key M
expect 3000 [MAP] menu: Main menu
wait 1500
shot keys-03-mainmenu.png
key BACK
wait 800
# back to Balanced for later scripts
serial pm1
expect 3000 [POWER] Balanced
quit
