# Top side button = lock (console "btn" presses it; the simulator's Device > Hold BOOT menu
# item holds / releases the real pin, GPIO0). Fresh device:
#   run-script.bat scripts\lock_button.txt lockbtn fresh
# then, on the same device name (no "fresh"), scripts\lock_button_boot.txt (a restart with a
# password: locked, on the idle screen) and scripts\lock_button_boot2.txt (a restart without
# one: the idle screen, not locked).
expect 60000 [BOOT] ready
serial wiz skip
wait 1500
# No password: the button shows the idle screen, nothing is locked.
serial btn
expect 3000 [BTN] top button (console): idle screen (no password)
expect 3000 [IDLE] enter
wait 1200
shot lockbtn-01-idle-nopass.png
serial lock
expect 3000 locked=0 password=0
key T
expect 3000 [IDLE] wake
wait 800
# The same with the real pin held low, then let go (the release does nothing).
menu Device > Hold BOOT
expect 3000 [BTN] top button (button): idle screen (no password)
expect 3000 [IDLE] enter
wait 800
menu Device > Hold BOOT
wait 800
serial lock
expect 3000 locked=0 password=0
key T
expect 3000 [IDLE] wake
wait 800
